<script>

    const question = document.getElementById("question");

    const yesBtn = document.getElementById("yesBtn");

    const noBtn = document.getElementById("noBtn");


    let yesSize = 20;


    const prompts = [

        "Are you sure? 🥺",

        "Really sure?? 💔",

        "Please say yes? 🙏",

        "I'll be very sad... 😭",

        "Pretty please? 🌹",

        "Okay fine... but Yes is still better 😌"

    ];


    let promptIndex = 0;


    noBtn.addEventListener("click", () => {

        // Change prompt

        question.innerText = prompts[promptIndex];

        promptIndex = (promptIndex + 1) % prompts.length;


        // Grow Yes button

        yesSize += 15;

        yesBtn.style.fontSize = yesSize + "px";

        yesBtn.style.padding = (yesSize / 2) + "px " + yesSize + "px";


        // Eventually make fullscreen

        if (yesSize > 200) {

            yesBtn.classList.add("fullscreen");

            yesBtn.innerText = "YES 💖";

            noBtn.style.display = "none";

            question.innerText = "You have no choice now 😌💘";

        }

    });


    yesBtn.addEventListener("click", () => {

        document.body.innerHTML = `

            <div style="

                color:white;

                text-align:center;

            ">

                YAY!!! 💖💖💖<br>

                Best decision ever 🥰

            </div>
